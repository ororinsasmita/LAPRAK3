#include <iostream>
using namespace std;

int main() {
    int angka[2][2];

    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            cin >> angka[i][j];
        }
    }

    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            cout << angka[i][j] << " ";
        }
        cout << endl;
    }

    return 0;
}
